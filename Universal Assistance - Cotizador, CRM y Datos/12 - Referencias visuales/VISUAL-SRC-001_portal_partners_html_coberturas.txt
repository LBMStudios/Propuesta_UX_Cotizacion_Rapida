        function showModalCobertura(producto) {
            var cobs = JSON.parse(producto.Atributos);
            $('#detBody').html('');
            var esAnual = producto.EsAnual;
            var svgColor = getSvgColor(esAnual);

            for (var i in cobs) {

                var exceptions = cobs[i].Exceptions;
                var tr = $('<tr class="accordion-header" style="position: relative;"/>');
                var tdInfo = "td-svg-" + i;
                var svgRowId = "div-svg-info-" + i;

                tr.append('<td style="padding-right: 2px;">' + cobs[i].NombreVisible + '</td>');
                tr.append('<td id="' + tdInfo + '" style="text-align:left;">' + cobs[i].ValorDisplay + '</td>');

                if (exceptions.length > 0) {

                    var row = getSvgRow(svgColor);
                    var info = getSvgInfo(svgColor, svgRowId);

                    var tdImg = $('<td class="tdImg-position-relative"></td>');
                    var divContainer = $('<div class="div-position-absolute"></div>');
                    divContainer.append(row);
                    tdImg.append(divContainer);

                    tr.append(tdImg);
                    $('#detBody').append(tr);

                    if (cobs[i].ValorDisplay !== " " && /\d/.test(cobs[i].ValorDisplay)) {
                        $('#' + tdInfo).append('  ').append(info);

                        $('#' + svgRowId).css({
                            "display": "inline-flex",
                            "align-items": "center",
                            "margin-left": "5px",
                            "position": "absolute"
                        });
                    }

                    row.css('cursor', 'pointer');

                    row.on('click', function () {
                        var $this = $(this);
                        var isExpanded = $this.hasClass("expanded");

                        if (isExpanded) {
                            $this.removeClass("expanded rotate");
                        } else {
                            $this.addClass("expanded rotate");
                        }

                        $(this).closest('tr').nextUntil('.accordion-header').toggle();
                        $(this).closest('tr').nextAll('#tr-vacio').first().hide();

                    });

                    for (var e in exceptions) {
                        var trVacio = $('<tr id="tr-vacio" class="accordion-content" style="display:none;"></tr>');
                        trVacio.append('<td colspan="3">&nbsp;</td>');

                        var trExcepcion = $('<tr id="tr-exep-' + e + '" class="accordion-content" style="display: none;"/>');
                        trExcepcion.append('<td id="td-exep-' + e + '" style="padding-left: 8px;font-size: x-small;font-style: italic;color: #6c757d;">• ' + exceptions[e].Descripcion + '</td>');
                        trExcepcion.append('<td id="td-exep-' + e + '" style="text-align:left;"> </td>');
                        trExcepcion.append('<td id="td-exep-' + e + '" style="text-align:left;"> </td>');

                        $('#detBody').append(trVacio);
                        $('#detBody').append(trExcepcion);
                    }

                } else {
                    $('#detBody').append(tr);
                }
            }
        }
